## 端口协商速率

```sh
log_conf --dup2pty

for i in 0 1 2 3 4; do
    STATUS=$(switch_ioctl --link_status $i 2>/dev/null)
    MODE=$(switch_ioctl --link_mode_get $i 2>/dev/null)
    echo "--------------------------------------------"
done

# =====================================================================
# 备注说明（速率代码对照表）：
#   mode = 3 : 100MF  (100M Full-Duplex / 百兆全双工)
#   mode = 5 : 1000MF (1000M Full-Duplex / 千兆全双工)
# =====================================================================

switch_test: link_status: port=0, status=0
switch_test: link_mode_get: port=0, mode=0, ret=-1
--------------------------------------------
switch_test: link_status: port=1, status=0
switch_test: link_mode_get: port=1, mode=0, ret=-1
--------------------------------------------
switch_test: link_status: port=2, status=0
switch_test: link_mode_get: port=2, mode=0, ret=-1
--------------------------------------------
switch_test: link_status: port=3, status=1
switch_test: link_mode_get: port=3, mode=5, ret=0
--------------------------------------------
switch_test: link_status: port=4, status=0
switch_test: link_mode_get: port=4, mode=0, ret=-1
--------------------------------------------
```
